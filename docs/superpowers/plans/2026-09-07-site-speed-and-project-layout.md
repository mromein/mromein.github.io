# Site Speed and Project Layout Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the home splash start playing within a second, cut every page's bytes by roughly 95%, and give project/design/experiment pages one clean, unified template with two credits-block variants for Matt to choose between.

**Architecture:** Jekyll 4.3 static site on GitHub Pages behind Cloudflare, no JS build step. One `project` layout composed of includes renders all three collections from a `p` variable (the page itself, or a `source` project for test pages). Media assets get lighter derivatives generated by shell scripts kept in `bin/`; originals move to an unpublished `_originals/` folder. The home splash streams one faststart mp4 into the existing PIXI mask pipeline instead of blob-fetching nine clips.

**Tech Stack:** Jekyll 4.3.4 / Liquid, SCSS (inlined per page via `scssify`), vanilla JS, PIXI 4.7 (home + info only), ffmpeg (encoding, derivatives), macOS `sips`/`mdls` for checks.

**Spec:** `docs/superpowers/specs/2026-09-07-site-speed-and-project-layout-design.md`

- [ ] **Step 3: Write the derivatives script**

Create `bin/derivatives.sh`:

```bash
#!/usr/bin/env bash
# Writes X-1600.jpg and X-800.jpg beside every documentation/preview JPG.
# Idempotent: regenerates only when the source is newer than the output.
# Never upscales (scale='min(W,iw)'). Keeps the embedded colour profile.
set -euo pipefail
cd "$(dirname "$0")/.."

find assets/01_projects assets/02_design assets/03_experiments -type f \
  \( -iname '*.jpg' -o -iname '*.jpeg' \) \
  ! -name '*-1600.jpg' ! -name '*-800.jpg' ! -name '*-poster.jpg' -print0 |
while IFS= read -r -d '' src; do
  base="${src%.*}"
  for w in 1600 800; do
    out="$base-$w.jpg"
    if [ ! -f "$out" ] || [ "$src" -nt "$out" ]; then
      ffmpeg -v error -y -i "$src" -vf "scale='min($w,iw)':-2" -q:v 4 "$out"
      echo "wrote $out"
    fi
  done
done
```

Run: `chmod +x bin/derivatives.sh`

- [ ] **Step 1: Write the conversion script**

Create `bin/gif-to-mp4.sh`:

```bash
#!/usr/bin/env bash
# Converts every documentation GIF under assets/ to a silent h264 mp4 loop plus a poster JPEG,
# then moves the GIF to _originals/ (same relative path). Safe to re-run: skips GIFs already moved.
set -euo pipefail
cd "$(dirname "$0")/.."

find assets/01_projects assets/02_design assets/03_experiments -type f -iname '*.gif' -print0 |
while IFS= read -r -d '' gif; do
  base="${gif%.*}"
  ffmpeg -v error -y -i "$gif" -an \
    -vf "scale=trunc(iw/2)*2:trunc(ih/2)*2,format=yuv420p" \
    -c:v libx264 -preset slow -crf 26 -movflags +faststart "$base.mp4"
  ffmpeg -v error -y -i "$gif" -frames:v 1 -q:v 5 "$base-poster.jpg"
  dest="_originals/$gif"
  mkdir -p "$(dirname "$dest")"
  mv "$gif" "$dest"
  printf "%s -> %s (%s bytes)\n" "$gif" "$base.mp4" "$(stat -f%z "$base.mp4")"
done
```

- [ ] **Step 1: Write the encode script**

Create `bin/encode-bg-videos.sh`:

```bash
#!/usr/bin/env bash
# Re-encodes the home splash clips: 1920x1080 h264, no audio, faststart, ~3.5 Mbps cap,
# plus a poster frame. Originals move to _originals/assets/bg_videos/.
set -euo pipefail
cd "$(dirname "$0")/.."
mkdir -p _originals/assets/bg_videos

for n in 01 02 03 04 05 06 07 08 09; do
  src="assets/bg_videos/$n.mp4"
  [ -f "$src" ] || src="_originals/assets/bg_videos/$n.mp4"
  ffmpeg -v error -y -i "$src" -an \
    -vf "scale=1920:1080:flags=lanczos" \
    -c:v libx264 -preset slow -crf 23 -maxrate 3500k -bufsize 7000k -profile:v high -level 4.1 \
    -pix_fmt yuv420p -g 60 -keyint_min 60 -sc_threshold 0 -movflags +faststart \
    "assets/bg_videos/$n-1080.mp4"
  ffmpeg -v error -y -ss 1 -i "$src" -frames:v 1 -vf scale=1280:720 -q:v 6 "assets/bg_videos/$n-poster.jpg"
  [ -f "assets/bg_videos/$n.mp4" ] && mv "assets/bg_videos/$n.mp4" "_originals/assets/bg_videos/$n.mp4"
  printf "%s: %s bytes\n" "$n-1080.mp4" "$(stat -f%z "assets/bg_videos/$n-1080.mp4")"
done
```

- [ ] **Step 3: Rewrite the home layout**

Replace `_layouts/home.html` with:

```html
<!DOCTYPE html>
<html lang="en">
{% capture style %}{% include home.scss %}{% endcapture %}
{% include head.html %}
<body>
  {% include header.html %}
  <video id="bg-video" muted playsinline autoplay preload="auto" aria-hidden="true"></video>
  <canvas id="bg-canvas"></canvas>
  <div id="loader"><p>Loading<span>.</span><span>.</span><span>.</span></p></div>

<script type="text/javascript">
(function(){
  var TOTAL = 9;
  var root = "{{ 'assets/bg_videos/' | absolute_url }}";
  var video = document.getElementById("bg-video");
  var canvas = document.getElementById("bg-canvas");
  var loader = document.getElementById("loader");
  var pixiTag = document.getElementById("pixi-script");

  function pad(n){ return (n < 10 ? "0" : "") + n; }
  function clipUrl(n){ return root + pad(n) + "-1080.mp4"; }
  function posterUrl(n){ return root + pad(n) + "-poster.jpg"; }

  var order = [];
  for (var i = 1; i <= TOTAL; i++) order.push(i);
  for (var j = order.length - 1; j > 0; j--) {
    var k = Math.floor(Math.random() * (j + 1));
    var t = order[j]; order[j] = order[k]; order[k] = t;
  }
  var pos = 0;

  // 1. Poster paints immediately.
  canvas.style.backgroundImage = "url(" + posterUrl(order[0]) + ")";

  var reduce = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  var saveData = navigator.connection && navigator.connection.saveData;
  if (reduce || saveData) { loader.style.display = "none"; return; }

  var pixiReady = !!window.PIXI, canPlay = false, pixiStarted = false;

  // 2. Stream the first clip straight from the network.
  video.src = clipUrl(order[0]);

  // 3. Prefetch the next clip as a blob while the current one plays.
  var nextBlob = null, fetching = false;
  function prefetchNext(){
    if (fetching || nextBlob) return;
    fetching = true;
    var n = order[(pos + 1) % TOTAL];
    var req = new XMLHttpRequest();
    req.open("GET", clipUrl(n), true);
    req.responseType = "blob";
    req.onload = function(){ if (req.status === 200) nextBlob = URL.createObjectURL(req.response); fetching = false; };
    req.onerror = function(){ fetching = false; };
    req.send();
  }

  video.addEventListener("playing", function(){
    loader.style.display = "none";
    if (!pixiStarted) canvas.style.backgroundImage = "none"; // reveal the raw video behind the canvas
    prefetchNext();
  });

  video.addEventListener("ended", function(){
    pos = (pos + 1) % TOTAL;
    if (nextBlob) {
      var url = nextBlob; nextBlob = null;
      video.src = url;
    } else {
      video.src = clipUrl(order[pos]);
    }
    video.play().catch(function(){});
    prefetchNext();
  });

  video.play().catch(function(){ /* autoplay refused: poster stays */ });

  // 4. PIXI compositing starts when both PIXI and the first frame are ready.
  function maybeStart(){
    if (pixiStarted || !pixiReady || !canPlay) return;
    pixiStarted = true;
    startPlayback();
  }
  if (pixiTag) {
    pixiTag.addEventListener("load", function(){ pixiReady = true; maybeStart(); });
    pixiTag.addEventListener("error", function(){ /* stay on the raw video */ });
  }
  video.addEventListener("canplay", function(){ canPlay = true; maybeStart(); });

  function startPlayback(){
    var pixiLoader = PIXI.loader;
    pixiLoader.add("largeMask", "{{ 'assets/large_mask.png' | absolute_url }}")
              .add("smallMask", "{{ 'assets/small_mask.png' | absolute_url }}");
    pixiLoader.load(function(_, resources){
      var ratio;
      function whichImage(){
        var tex = window.innerWidth > 1024 ? resources.largeMask.texture : resources.smallMask.texture;
        ratio = tex.height / tex.width;
        return tex;
      }

      var renderer = PIXI.autoDetectRenderer(window.innerWidth, window.innerHeight, {
        antialias: true, view: canvas, autoResize: true
      });
      canvas.style.backgroundImage = "none";
      // The raw <video> stays visible behind the (opaque) canvas: hiding it can stall decoding in Safari.

      var stage = new PIXI.Container();
      var texture = PIXI.Texture.fromVideo(video);

      var bg = new PIXI.Sprite(texture);
      bg.anchor.set(0.5, 0.5);
      stage.addChild(bg);

      var bgFront = new PIXI.Sprite(texture);
      bgFront.anchor.set(0.5, 0.5);
      stage.addChild(bgFront);

      var thing = new PIXI.Sprite(whichImage());
      thing.anchor.set(0.5, 0.5);
      bgFront.mask = thing;
      stage.addChild(thing);

      function resize(){
        var width = window.innerWidth, height = window.innerHeight;
        renderer.resize(width, height);
        var tw = texture.width || 1920, th = texture.height || 1080;
        var bgratio = th / tw;
        if (tw / th <= width / height) {
          bg.width = bgFront.width = width;
          bg.height = bgFront.height = width * bgratio;
        } else {
          bg.height = bgFront.height = height;
          bg.width = bgFront.width = height / bgratio;
        }
        thing.texture = whichImage();
        thing.width = width * 0.8;
        thing.height = ratio * thing.width;
        thing.position.set(width / 2, height / 2);
        bgFront.position.set(width / 2, height / 1.2);
        bg.position.set(width / 2, height / 2);
      }
      resize();
      window.addEventListener("resize", resize, false);
      window.addEventListener("orientationchange", resize, false);

      (function animate(){ renderer.render(stage); requestAnimationFrame(animate); })();
    });
  }
})();
</script>
</body>
</html>
```
